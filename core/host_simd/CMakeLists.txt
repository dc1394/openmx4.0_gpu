cmake_minimum_required(VERSION 3.25.2)
project(OpenMXHostSIMD LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
set(CMAKE_VISIBILITY_INLINES_HIDDEN ON)
include(CheckCXXSourceCompiles)
check_cxx_source_compiles("
  #if defined(__CUDACC__) || defined(__HIPCC__) || defined(__SYCL_DEVICE_ONLY__)
  #error Device compiler selected for host SIMD
  #endif
  int main() { return 0; }" OPENMX_SIMD_ORDINARY_CXX)
if(NOT OPENMX_SIMD_ORDINARY_CXX)
  message(FATAL_ERROR "Set OPENMX_HOST_SIMD_CXX_COMPILER to an ordinary C++20 compiler (e.g. g++ or clang++), not a device compiler wrapper")
endif()
# This is an independent configuration of the SAME pinned Kokkos sources.
# No backend/architecture macros, compiler flags, or symbols are shared with
# the application's Kokkos. No Kokkos object crosses our two-function C ABI.
set(Kokkos_ENABLE_SERIAL ON CACHE BOOL "" FORCE)
foreach(device CUDA HIP SYCL OPENMP THREADS OPENACC HPX)
  set(Kokkos_ENABLE_${device} OFF CACHE BOOL "" FORCE)
endforeach()
set(Kokkos_ENABLE_TESTS OFF CACHE BOOL "" FORCE)
set(Kokkos_ENABLE_EXAMPLES OFF CACHE BOOL "" FORCE)
set(Kokkos_ENABLE_BENCHMARKS OFF CACHE BOOL "" FORCE)
set(Kokkos_ENABLE_COMPILE_AS_CMAKE_LANGUAGE OFF CACHE BOOL "" FORCE)
add_subdirectory("${OPENMX_KOKKOS_SOURCE_DIR}" kokkos EXCLUDE_FROM_ALL)
include(CheckCXXCompilerFlag)
add_library(openmx_host_simd SHARED)
include("${CMAKE_CURRENT_SOURCE_DIR}/../../cmake/OpenMXCompilerRuntime.cmake")
openmx_compiler_runtime_rpath(openmx_host_simd CXX)
set_target_properties(openmx_host_simd PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
foreach(config DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
  set_target_properties(openmx_host_simd PROPERTIES LIBRARY_OUTPUT_DIRECTORY_${config} "${CMAKE_BINARY_DIR}/lib")
endforeach()
target_link_libraries(openmx_host_simd PRIVATE Kokkos::kokkos)
if(UNIX AND NOT APPLE)
  # Keep the private SERIAL Kokkos symbols out of the application's global
  # symbol namespace, including debug-mode assertion support pulled from .a.
  target_link_options(openmx_host_simd PRIVATE "LINKER:--exclude-libs,ALL")
endif()
if(OPENMX_CORE_SANITIZERS)
  target_link_options(openmx_host_simd PRIVATE -fsanitize=address,undefined)
endif()
function(openmx_host_object name)
  add_library(${name} OBJECT ${ARGN})
  target_include_directories(${name} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../include")
  target_link_libraries(${name} PRIVATE Kokkos::kokkos)
  target_compile_definitions(${name} PRIVATE OPENMX_HOST_SIMD_BUILD=1)
  target_compile_features(${name} PRIVATE cxx_std_20)
  set_target_properties(${name} PROPERTIES POSITION_INDEPENDENT_CODE ON
    CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON)
  target_compile_options(${name} PRIVATE
    "$<$<CXX_COMPILER_ID:GNU,Clang,AppleClang>:-fno-fast-math;-ffp-contract=off>"
    "$<$<CXX_COMPILER_ID:IntelLLVM>:-fp-model=precise;-fno-fast-math;-ffp-contract=off>")
  if(OPENMX_CORE_SANITIZERS)
    target_compile_options(${name} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  endif()
  target_sources(openmx_host_simd PRIVATE $<TARGET_OBJECTS:${name}>)
endfunction()

openmx_host_object(openmx_host_grid_dispatch ../src/host_grid_dispatch.cpp)
openmx_host_object(openmx_host_grid_scalar ../src/host_grid_simd.cpp)
if(OPENMX_CORE_HOST_SIMD AND CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|AMD64|amd64|i.86)$"
    AND CMAKE_CXX_COMPILER_ID MATCHES "^(GNU|Clang|AppleClang|IntelLLVM)$")
  check_cxx_compiler_flag("-mavx2" OPENMX_HOST_HAS_AVX2)
  check_cxx_compiler_flag("-mavx512f -mavx512dq -mavx512vl" OPENMX_HOST_HAS_AVX512)
  if(OPENMX_HOST_HAS_AVX2)
    openmx_host_object(openmx_host_grid_avx2 ../src/host_grid_simd.cpp)
    target_compile_options(openmx_host_grid_avx2 PRIVATE -mavx2)
    target_compile_definitions(openmx_host_grid_avx2 PRIVATE OPENMX_HOST_SIMD_AVX2=1)
    target_compile_definitions(openmx_host_grid_dispatch PRIVATE OPENMX_HOST_SIMD_AVX2=1)
  endif()
  if(OPENMX_HOST_HAS_AVX512)
    openmx_host_object(openmx_host_grid_avx512 ../src/host_grid_simd.cpp)
    target_compile_options(openmx_host_grid_avx512 PRIVATE -mavx512f -mavx512dq -mavx512vl)
    target_compile_definitions(openmx_host_grid_avx512 PRIVATE OPENMX_HOST_SIMD_AVX512=1)
    target_compile_definitions(openmx_host_grid_dispatch PRIVATE OPENMX_HOST_SIMD_AVX512=1)
  endif()
elseif(OPENMX_CORE_HOST_SIMD AND CMAKE_SYSTEM_PROCESSOR MATCHES "^(aarch64|arm64|ARM64)$")
  # AArch64, including Apple Silicon, guarantees FP64 Advanced SIMD.
  openmx_host_object(openmx_host_grid_neon ../src/host_grid_simd.cpp)
  target_compile_definitions(openmx_host_grid_neon PRIVATE OPENMX_HOST_SIMD_NEON=1)
  target_compile_definitions(openmx_host_grid_dispatch PRIVATE OPENMX_HOST_SIMD_NEON=1)
endif()
